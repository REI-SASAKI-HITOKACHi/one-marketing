function apiCreateEstimate(payload) {
  return withApi_('見積作成', '', function () {
    setupInitialSheets();

    const ctx = loadContext_();
    const templateSs = getTemplateSs_(ctx.settings);
    const dataSheet = ensureSheetWithHeaders_(templateSs, APP.SHEET_ESTIMATE_DATA, getEstimateHeaders_());

    const lock = LockService.getScriptLock();
    lock.waitLock(30000);

    let estimateId = '';
    let record = null;
    let calc = null;

    try {
      estimateId = generateEstimateId_(dataSheet);
      calc = calculateEstimate_(payload, ctx);
      record = buildEstimateRecord_(payload, ctx, calc, estimateId);
      appendObject_(dataSheet, getEstimateHeaders_(), record);
      log_('見積作成', estimateId, '見積データを保存しました。', '');
    } finally {
      lock.releaseLock();
    }

    const errors = [];
    let pdfFile = null;
    let pdfUrl = '';
    let pdfFileId = '';

    try {
      const pdfResult = generateEstimatePdf_(record, calc, ctx);
      pdfFile = pdfResult.file;
      pdfUrl = pdfResult.url;
      pdfFileId = pdfResult.fileId;

      updateEstimateFields_(estimateId, {
        PDF_URL: pdfUrl,
        PDF_FILE_ID: pdfFileId,
        更新日時: new Date()
      });

      record.PDF_URL = pdfUrl;
      record.PDF_FILE_ID = pdfFileId;
      log_('PDF生成', estimateId, 'PDFを生成しました。', '');
    } catch (e) {
      const msg = toErrorMessage_(e);
      errors.push('PDF生成失敗：' + msg);
      log_('エラー', estimateId, 'PDF生成に失敗しました。', msg);
    }

    let draftUrl = '';
    let draftId = '';
    let mailPreview = null;

    try {
      if (!pdfFile) {
        throw new Error('PDF生成に失敗したため、PDF添付済みメール下書きは作成していません。');
      }

      const draftResult = createEstimateMailDraft_(record, calc, ctx, pdfFile);
      draftUrl = draftResult.draftUrl;
      draftId = draftResult.draftId;

      updateEstimateFields_(estimateId, {
        メール下書きURL: draftUrl,
        メール下書きID: draftId,
        メール下書き作成日時: new Date(),
        更新日時: new Date()
      });

      log_('メール下書き作成', estimateId, 'Gmail下書きを作成しました。', '');
    } catch (e) {
      const msg = toErrorMessage_(e);
      errors.push('メール下書き作成失敗：' + msg);
      mailPreview = buildEstimateMailPreview_(record, calc, ctx);
      log_('エラー', estimateId, 'メール下書き作成に失敗しました。', msg);
    }

    return {
      ok: true,
      estimateId: estimateId,
      calc: calc,
      pdfUrl: pdfUrl,
      pdfFileId: pdfFileId,
      draftUrl: draftUrl,
      draftId: draftId,
      errors: errors,
      mailPreview: mailPreview,
      representativeDraftAvailable: calc.exceptionFlag === true
    };
  });
}

function apiCreateRepresentativeDraft(estimateId) {
  return withApi_('代表者確認メール下書き作成', estimateId, function () {
    const ctx = loadContext_();
    const representativeEmail = String(ctx.settings['代表者メール'] || '').trim();

    if (!representativeEmail) {
      throw new Error('設定マスタの representative_email または 代表者メール が未設定です。');
    }

    const templateSs = getTemplateSs_(ctx.settings);
    const dataSheet = ensureSheetWithHeaders_(templateSs, APP.SHEET_ESTIMATE_DATA, getEstimateHeaders_());
    const found = findEstimateRecord_(dataSheet, estimateId);
    if (!found) throw new Error('見積IDが見つかりません：' + estimateId);

    const record = found.record;
    const template = ctx.mailTemplates['代表者確認依頼'] || getFallbackRepresentativeTemplate_();

    const tokens = buildTokensFromRecord_(record, {
      例外理由: record['例外理由'] || '',
      PDF_URL: record['PDF_URL'] || '',
      作成日時: formatDateTime_(record['作成日時'] || new Date())
    });

    const options = buildGmailOptions_(record['送信元メール'], '', ctx.settings);
    const draft = GmailApp.createDraft(
      representativeEmail,
      renderTemplate_(template.subject, tokens),
      renderTemplate_(template.body, tokens),
      options
    );

    const draftId = draft.getId();
    const draftUrl = getGmailDraftsUrl_();

    updateEstimateFields_(estimateId, {
      代表者確認メール下書きURL: draftUrl,
      代表者確認メール下書きID: draftId,
      代表者確認メール作成日時: new Date(),
      更新日時: new Date()
    });

    log_('代表者確認メール下書き作成', estimateId, '代表者確認メール下書きを作成しました。', '');
    return { ok: true, draftId: draftId, draftUrl: draftUrl };
  });
}

function apiSearchEstimates(criteria) {
  return withApi_('過去見積検索', '', function () {
    const ctx = loadContext_();
    const templateSs = getTemplateSs_(ctx.settings);
    const dataSheet = ensureSheetWithHeaders_(templateSs, APP.SHEET_ESTIMATE_DATA, getEstimateHeaders_());
    const rows = readObjects_(dataSheet, ['estimate_id']);
    const c = criteria || {};

    const results = rows
      .filter(function (r) {
        return matchText_(r['estimate_id'], c.estimateId)
          && matchText_(r['顧客名'], c.customerName)
          && matchText_(r['案件名'], c.projectName)
          && matchText_(r['案件タイプ'], c.projectType)
          && matchDate_(r['作成日時'], c.createdDate);
      })
      .sort(function (a, b) {
        return new Date(b['作成日時']).getTime() - new Date(a['作成日時']).getTime();
      })
      .slice(0, 50)
      .map(function (r) {
        return {
          estimate_id: r['estimate_id'] || '',
          created_at: formatDateTime_(r['作成日時']),
          customer_name: r['顧客名'] || '',
          project_name: r['案件名'] || '',
          project_type: r['案件タイプ'] || '',
          total_amount: toNumber_(r['合計金額']),
          total_amount_display: formatYen_(toNumber_(r['合計金額'])),
          pdf_url: r['PDF_URL'] || '',
          exception_flag: String(r['例外フラグ'] || '') === 'TRUE'
        };
      });

    log_('過去見積呼び出し', '', '検索件数：' + results.length, '');
    return { ok: true, results: results };
  });
}

/* ========================= 計算 ========================= */

function calculateEstimate_(payload, ctx) {
  const p = payload || {};
  const settings = ctx.settings;
  const menuMap = ctx.menuMap;
  const rawDetails = Array.isArray(p.details) ? p.details : [];
  const lines = [];

  rawDetails.forEach(function (d) {
    if (!d) return;
    const menuId = String(d.menuId || '').trim();
    if (!menuId) return;

    const menu = menuMap[menuId] || null;
    const qty = Math.max(0, toNumber_(d.qty || 0));
    const unitPriceMissing = !menu || isBlank_(menu.unitPriceRaw);
    const unitPrice = menu ? toNumber_(menu.unitPrice) : 0;
    const amount = roundYen_(qty * unitPrice);

    lines.push({
      menuId: menuId,
      name: menu ? menu.name : String(d.menuName || ''),
      menuType: menu ? menu.menuType : String(d.menuType || ''),
      rawMenuType: menu ? menu.rawMenuType : String(d.menuType || ''),
      category: menu ? menu.category : '',
      qty: qty,
      unit: menu ? menu.unit : '',
      unitPrice: unitPrice,
      unitPriceMissing: unitPriceMissing,
      amount: amount,
      taxType: menu ? menu.taxType : '課税',
      busyTarget: menu ? menu.busyTarget : true,
      busySurchargeRaw: menu ? menu.busySurchargeRaw : '',
      discountTarget: menu ? menu.discountTarget : true,
      multipleDiscountTarget: menu ? menu.multipleDiscountTarget : false,
      invoiceReusable: menu ? menu.invoiceReusable : true,
      note: menu ? menu.note : '',
      requireCheck: menu ? menu.requireCheck : ''
    });
  });

  const workDate = parseDateInput_(p.workDate) || new Date();
  const workMonth = Number(formatDate_(workDate, 'M'));
  const taxRate = getTaxRate_(settings);
  const highwayFee = Math.max(0, toNumber_(p.highwayFee || 0));

  const lineSubtotal = lines.reduce(function (sum, l) { return sum + l.amount; }, 0);
  const taxableLineSubtotal = lines
    .filter(function (l) { return isTaxable_(l.taxType); })
    .reduce(function (sum, l) { return sum + l.amount; }, 0);
  const nonTaxLineSubtotal = lines
    .filter(function (l) { return !isTaxable_(l.taxType); })
    .reduce(function (sum, l) { return sum + l.amount; }, 0);

  const busyAuto = judgeBusySeason_(workMonth, lines, p, ctx);
  const busyManual = p.busyManual === null || p.busyManual === undefined || p.busyManual === ''
    ? busyAuto
    : parseBooleanLoose_(p.busyManual);

  const busyAmount = busyManual ? calcBusyAmount_(lines, p, ctx) : 0;
  const busyPdfRow = busyManual ? buildBusyPdfRow_(lines, p, ctx, busyAmount) : null;

  const discountInfo = calcDiscountInfo_(workMonth, lines, ctx);
  const discountAuto = discountInfo.auto;
  const discountManual = p.discountManual === null || p.discountManual === undefined || p.discountManual === ''
    ? discountAuto
    : parseBooleanLoose_(p.discountManual);
  const discountAmount = discountManual ? discountInfo.amount : 0;

  const taxableBeforeDiscount = taxableLineSubtotal + busyAmount;
  const taxableDiscount = Math.min(discountAmount, taxableBeforeDiscount);
  const taxableSubtotal = Math.max(0, taxableBeforeDiscount - taxableDiscount);
  const nonTaxableSubtotal = nonTaxLineSubtotal + highwayFee;
  const tax = roundYen_(taxableSubtotal * taxRate);
  const grandTotal = taxableSubtotal + nonTaxableSubtotal + tax;

  const pdfRows = buildPdfDetailRowsFromCalc_({
    lines: lines,
    busyAmount: busyAmount,
    busyPdfRow: busyPdfRow,
    discountAmount: discountAmount,
    highwayFee: highwayFee
  });

  const exceptions = [];
  if (grandTotal === 0) exceptions.push('合計金額が0円です。');
  if (lines.some(function (l) { return l.unitPriceMissing; })) exceptions.push('単価未設定メニューが含まれています。');

  lines.forEach(function (l) {
    if (l.requireCheck) exceptions.push('要確認事項あり：' + l.name + '（' + l.requireCheck + '）');
  });

  const totalQtyForMultiple = lines
    .filter(function (l) { return l.multipleDiscountTarget; })
    .reduce(function (sum, l) { return sum + l.qty; }, 0);

  if (totalQtyForMultiple > 50) exceptions.push('複数台割引対象メニューの合計台数が50台を超えています。');
  if (lines.some(function (l) { return containsAny_(l.rawMenuType + ' ' + l.category + ' ' + l.name, ['空室']); })) exceptions.push('空室清掃メニューが含まれています。');
  if (lines.some(function (l) { return containsAny_(l.rawMenuType + ' ' + l.category + ' ' + l.name, ['法人', '事業者']); }) || containsAny_(p.projectType, ['法人', '事業者'])) exceptions.push('法人/事業者案件または法人/事業者メニューが含まれています。');
  if (busyManual !== busyAuto || discountManual !== discountAuto) exceptions.push('繁忙期または割引の手動設定が自動判定と異なります。');
  if (highwayFee > 0) exceptions.push('高速代が入力されています。');
  if (String(p.remarks || '').indexOf('要確認') >= 0) exceptions.push('備考に「要確認」が含まれています。');
  if (lines.length > APP.MAX_DETAIL_ROWS) exceptions.push('保存明細が16行を超えています。');
  if (pdfRows.length > APP.MAX_DETAIL_ROWS) exceptions.push('PDF表示用の明細行が16行を超えています。');

  return {
    lines: lines,
    pdfRows: pdfRows,
    lineSubtotal: lineSubtotal,
    mainTotal: lines.filter(function (l) { return l.menuType === 'メイン'; }).reduce(function (s, l) { return s + l.amount; }, 0),
    optionTotal: lines.filter(function (l) { return l.menuType === 'オプション'; }).reduce(function (s, l) { return s + l.amount; }, 0),
    highwayFee: highwayFee,
    busyAuto: busyAuto,
    busyManual: busyManual,
    busyAmount: busyAmount,
    discountAuto: discountAuto,
    discountManual: discountManual,
    discountAmount: discountAmount,
    discountType: discountInfo.type,
    taxableSubtotal: taxableSubtotal,
    nonTaxableSubtotal: nonTaxableSubtotal,
    taxRate: taxRate,
    tax: tax,
    grandTotal: grandTotal,
    exceptionFlag: exceptions.length > 0,
    exceptionReasons: exceptions,
    summaryDisplay: {
      lineSubtotal: formatYen_(lineSubtotal),
      busyAmount: formatYen_(busyAmount),
      discountAmount: formatYen_(discountAmount),
      highwayFee: formatYen_(highwayFee),
      taxableSubtotal: formatYen_(taxableSubtotal),
      nonTaxableSubtotal: formatYen_(nonTaxableSubtotal),
      tax: formatYen_(tax),
      grandTotal: formatYen_(grandTotal)
    }
  };
}

function getBusyTargetLines_(lines) {
  return lines.filter(function (l) {
    return l.menuType === 'メイン' && l.busyTarget !== false;
  });
}

function judgeBusySeason_(workMonth, lines, payload, ctx) {
  const rule = findMonthRule_(ctx.discountRules, '繁忙期', workMonth);
  if (!rule) return false;

  const mainLines = getBusyTargetLines_(lines);
  if (mainLines.length === 0) return false;

  const totalQty = mainLines.reduce(function (sum, l) { return sum + l.qty; }, 0);
  const hasCorporate = lines.some(function (l) {
    return containsAny_(l.rawMenuType + ' ' + l.category + ' ' + l.name, ['法人', '事業者']);
  }) || containsAny_(payload.projectType, ['法人', '事業者']);

  if (hasCorporate && totalQty > 5) return false;
  return true;
}

function calcBusyAmount_(lines, payload, ctx) {
  const settings = ctx.settings;
  const defaultAmount = toNumber_(settings['繁忙期加算額'] || 3300);
  const unitType = String(settings['繁忙期加算単位'] || '数量ごと').trim();
  const targetLines = getBusyTargetLines_(lines);

  if (targetLines.length === 0) return 0;
  if (unitType === '見積ごと') return defaultAmount;

  if (unitType === '明細ごと') {
    return targetLines.reduce(function (sum, l) {
      const raw = isBlank_(l.busySurchargeRaw) ? defaultAmount : l.busySurchargeRaw;
      return sum + calcBusySurchargeForLine_(l, raw, false);
    }, 0);
  }

  return targetLines.reduce(function (sum, l) {
    const raw = isBlank_(l.busySurchargeRaw) ? defaultAmount : l.busySurchargeRaw;
    return sum + calcBusySurchargeForLine_(l, raw, true);
  }, 0);
}

function buildBusyPdfRow_(lines, payload, ctx, busyAmount) {
  if (!busyAmount) return null;

  const settings = ctx.settings;
  const defaultAmount = toNumber_(settings['繁忙期加算額'] || 3300);
  const unitType = String(settings['繁忙期加算単位'] || '数量ごと').trim();
  const targetLines = getBusyTargetLines_(lines);

  if (targetLines.length === 0) return null;

  if (unitType !== '数量ごと') {
    return { name: '繁忙期加算', qty: 1, unitPrice: busyAmount, amount: busyAmount, taxType: '課税' };
  }

  const totalQty = targetLines.reduce(function (sum, l) { return sum + toNumber_(l.qty); }, 0);
  const surchargeValues = targetLines.map(function (l) {
    const raw = isBlank_(l.busySurchargeRaw) ? defaultAmount : l.busySurchargeRaw;
    const rawText = String(raw || '').trim();

    if (rawText.indexOf('%') >= 0) {
      return { type: 'rate', raw: rawText, value: normalizeRate_(rawText) };
    }

    return { type: 'amount', raw: rawText, value: toNumber_(rawText) };
  });

  const allFixedAmount = surchargeValues.length > 0 && surchargeValues.every(function (x) {
    return x.type === 'amount' && x.value === surchargeValues[0].value;
  });

  if (totalQty > 0 && allFixedAmount) {
    return {
      name: '繁忙期加算',
      qty: totalQty,
      unitPrice: surchargeValues[0].value,
      amount: busyAmount,
      taxType: '課税'
    };
  }

  return { name: '繁忙期加算', qty: 1, unitPrice: busyAmount, amount: busyAmount, taxType: '課税' };
}

function calcBusySurchargeForLine_(line, raw, multiplyByQty) {
  const rawText = String(raw || '').trim();
  if (!rawText) return 0;

  if (rawText.indexOf('%') >= 0) {
    return roundYen_(line.amount * normalizeRate_(rawText));
  }

  const amount = toNumber_(rawText);
  return multiplyByQty ? roundYen_(amount * line.qty) : roundYen_(amount);
}

function calcDiscountInfo_(workMonth, lines, ctx) {
  const earlyRule = findMonthRule_(ctx.discountRules, '早期予約割引', workMonth);

  const earlyEligibleSubtotal = lines
    .filter(function (l) { return l.discountTarget !== false; })
    .reduce(function (sum, l) { return sum + l.amount; }, 0);

  const totalQty = lines
    .filter(function (l) { return l.multipleDiscountTarget; })
    .reduce(function (sum, l) { return sum + l.qty; }, 0);

  if (earlyRule && earlyEligibleSubtotal > 0) {
    return {
      auto: true,
      type: '早期予約割引',
      amount: roundYen_(earlyEligibleSubtotal * normalizeRate_(earlyRule.value))
    };
  }

  if (totalQty >= 2) {
    let amount = 0;

    lines.forEach(function (l) {
      if (!l.multipleDiscountTarget) return;
      const category = classifyAirconDiscountCategory_(l);
      const rule = findMultipleDiscountRule_(ctx.discountRules, category, totalQty);
      if (rule) amount += toNumber_(rule.value) * l.qty;
    });

    return { auto: amount > 0, type: amount > 0 ? '複数台割引' : '', amount: roundYen_(amount) };
  }

  return { auto: false, type: '', amount: 0 };
}

function findMonthRule_(rules, ruleType, month) {
  const matched = rules
    .filter(function (r) {
      if (r.ruleType !== ruleType) return false;
      const start = toNumber_(r.startMonth);
      const end = toNumber_(r.endMonth);
      if (!start || !end) return false;
      if (start <= end) return month >= start && month <= end;
      return month >= start || month <= end;
    })
    .sort(function (a, b) { return toNumber_(a.priority) - toNumber_(b.priority); });

  return matched[0] || null;
}

function findMultipleDiscountRule_(rules, target, totalQty) {
  const matched = rules
    .filter(function (r) {
      if (r.ruleType !== '複数台割引') return false;
      if (r.target !== target) return false;
      return matchQtyCondition_(r.condition, totalQty);
    })
    .sort(function (a, b) { return toNumber_(a.priority) - toNumber_(b.priority); });

  return matched[0] || null;
}

function matchQtyCondition_(condition, qty) {
  const c = String(condition || '');
  const rangeMatch = c.match(/(\d+)\s*-\s*(\d+)/);
  if (rangeMatch) return qty >= Number(rangeMatch[1]) && qty <= Number(rangeMatch[2]);

  const overMatch = c.match(/(\d+)\s*超/);
  if (overMatch) return qty > Number(overMatch[1]);

  return false;
}

function classifyAirconDiscountCategory_(line) {
  const text = String((line.category || '') + ' ' + (line.name || ''));
  if (containsAny_(text, ['ロボ', 'お掃除機能', '掃除機能'])) return 'ロボ付きエアコン';
  if (containsAny_(text, ['業務用', '天カセ', '天井', 'パッケージ'])) return '業務用エアコン';
  return 'ノーマルエアコン';
}

function resolveSubmitTarget_(payload, ctx) {
  const p = payload || {};
  const projectType = String(p.projectType || '').trim();
  const target = (ctx.submitTargets || []).find(function (t) {
    return t.projectType === projectType;
  }) || null;

  if (projectType === '自社') {
    return {
      submitToName: '自社',
      addresseeName: p.customerName || '',
      addresseeSuffix: p.addresseeSuffix || '様',
      to: p.selfRecipientEmail || '',
      cc: target ? target.cc : (ctx.settings['固定CC'] || ''),
      from: target ? (target.from || ctx.settings['送信元メール']) : (ctx.settings['送信元メール'] || APP.DEFAULT_EXECUTION_EMAIL),
      templateType: target ? (target.templateType || '自社案件_見積送付') : '自社案件_見積送付'
    };
  }

  if (!target) {
    return {
      submitToName: projectType,
      addresseeName: projectType,
      addresseeSuffix: '御中',
      to: '',
      cc: ctx.settings['固定CC'] || '',
      from: ctx.settings['送信元メール'] || APP.DEFAULT_EXECUTION_EMAIL,
      templateType: '元請案件_見積送付'
    };
  }

  return {
    submitToName: target.submitToName || projectType,
    addresseeName: target.addresseeName || target.submitToName || projectType,
    addresseeSuffix: target.addresseeName ? '' : '御中',
    to: target.to || '',
    cc: target.cc || '',
    from: target.from || ctx.settings['送信元メール'] || APP.DEFAULT_EXECUTION_EMAIL,
    templateType: target.templateType || '元請案件_見積送付'
  };
}

function generateEstimatePdf_(record, calc, ctx) {
  const templateId = ctx.settings['見積請求書テンプレートスプレッドシートID'] || APP.DEFAULT_TEMPLATE_SS_ID;
  const folderId = ctx.settings['見積書PDF保存フォルダID'] || ctx.settings['PDF保存フォルダID'] || APP.DEFAULT_ESTIMATE_FOLDER_ID;

  if (!templateId) throw new Error('見積請求書テンプレートスプレッドシートIDが未設定です。');
  if (!folderId) throw new Error('見積書PDF保存フォルダIDが未設定です。');

  const folder = DriveApp.getFolderById(folderId);
  const sourceFile = DriveApp.getFileById(templateId);
  const tmpFile = sourceFile.makeCopy('tmp_' + record.estimate_id + '_' + Date.now());
  const tmpSs = SpreadsheetApp.openById(tmpFile.getId());

  try {
    const cellDefs = getCellDefMap_('見積書', ctx);
    const targetSheetName = getTargetSheetNameFromDefs_(cellDefs, APP.TEMP_ESTIMATE_SHEET);
    const sheet = tmpSs.getSheetByName(targetSheetName);
    if (!sheet) throw new Error('対象シートが見つかりません：' + targetSheetName);

    fillEstimateSheet_(sheet, cellDefs, record, calc, ctx.settings);
    SpreadsheetApp.flush();
    Utilities.sleep(700);

    const pdfBlob = exportSheetToPdfBlob_(tmpSs.getId(), sheet.getSheetId());
    const fileName = sanitizeFileName_(
      record.estimate_id + '_見積書_' + record['顧客名'] + '_' + record['案件名'] + '.pdf'
    );

    pdfBlob.setName(fileName);
    const savedFile = folder.createFile(pdfBlob);

    return { file: savedFile, url: savedFile.getUrl(), fileId: savedFile.getId() };
  } finally {
    tmpFile.setTrashed(true);
  }
}

function writeDetailRows_(sheet, defs, rows) {
  const nameDef = defs['detail_name_range'];
  const qtyDef = defs['detail_qty_range'];
  const unitDef = defs['detail_unit_price_range'];
  const amountDef = defs['detail_amount_range'];

  if (!nameDef || !qtyDef || !unitDef || !amountDef) {
    throw new Error('差し込みセル定義に明細範囲が不足しています。');
  }

  const nameRange = sheet.getRange(nameDef.a1);
  const qtyRange = sheet.getRange(qtyDef.a1);
  const unitRange = sheet.getRange(unitDef.a1);
  const amountRange = sheet.getRange(amountDef.a1);

  nameRange.clearContent();
  qtyRange.clearContent();
  unitRange.clearContent();
  amountRange.clearContent();

  const maxRows = Math.min(APP.MAX_DETAIL_ROWS, nameRange.getNumRows());

  for (let i = 0; i < maxRows; i++) {
    const row = rows[i] || {};
    sheet.getRange(nameRange.getRow() + i, nameRange.getColumn()).setValue(row.name || '');
    sheet.getRange(qtyRange.getRow() + i, qtyRange.getColumn()).setValue(row.qty || '');
    sheet.getRange(unitRange.getRow() + i, unitRange.getColumn()).setValue(row.unitPrice === '' || row.unitPrice === undefined ? '' : row.unitPrice);
    sheet.getRange(amountRange.getRow() + i, amountRange.getColumn()).setValue(row.amount === '' || row.amount === undefined ? '' : row.amount);
  }
}

function exportSheetToPdfBlob_(spreadsheetId, gid) {
  const url = 'https://docs.google.com/spreadsheets/d/' + encodeURIComponent(spreadsheetId) + '/export'
    + '?format=pdf'
    + '&gid=' + encodeURIComponent(gid)
    + '&size=A4'
    + '&portrait=true'
    + '&fitw=true'
    + '&sheetnames=false'
    + '&printtitle=false'
    + '&pagenumbers=false'
    + '&gridlines=false'
    + '&fzr=false';

  const res = UrlFetchApp.fetch(url, {
    headers: { Authorization: 'Bearer ' + ScriptApp.getOAuthToken() },
    muteHttpExceptions: true
  });

  if (res.getResponseCode() !== 200) {
    throw new Error('PDFエクスポートに失敗しました。HTTP ' + res.getResponseCode() + ' / ' + res.getContentText().slice(0, 300));
  }

  return res.getBlob().setContentType(MimeType.PDF);
}

  const valueHeader = info.map['設定値'] ? '設定値' : '値';

function getMenus_() {
  const sheet = ensureSheetWithHeaders_(getMasterSs_(), APP.SHEET_MENU, getMenuHeaders_());
  const headerInfo = findHeaderInfo_(sheet, ['メニューID', 'メニュー名']);
  if (!headerInfo || !headerInfo.headerRow) return [];

  const lastRow = sheet.getLastRow();
  const lastCol = sheet.getLastColumn();
  if (lastRow <= headerInfo.headerRow) return [];

  const headers = sheet.getRange(headerInfo.headerRow, 1, 1, lastCol).getDisplayValues()[0]
    .map(function (h) { return String(h || '').trim(); });
  const values = sheet.getRange(headerInfo.headerRow + 1, 1, lastRow - headerInfo.headerRow, lastCol).getValues();

  return values
    .filter(function (row) { return row.some(function (v) { return !isBlank_(v); }); })
    .map(function (row) {
      const menuId = String(pickRowValue_(headers, row, ['メニューID']) || '').trim();
      const rawMenuType = String(pickRowValue_(headers, row, ['メニュータイプ']) || '').trim();

      return {
        menuId: menuId,
        name: String(pickRowValue_(headers, row, ['顧客表示名', 'メニュー名']) || '').trim(),
        rawMenuType: rawMenuType,
        menuType: normalizeMenuType_(menuId, rawMenuType),
        category: String(pickRowValue_(headers, row, ['カテゴリ']) || '').trim(),
        unitPriceRaw: pickRowValue_(headers, row, ['単価', '価格']),
        unitPrice: toNumber_(pickRowValue_(headers, row, ['単価', '価格'])),
        taxType: String(pickRowValue_(headers, row, ['税区分']) || '課税').trim(),
        standardTime: pickRowValue_(headers, row, ['標準時間_分', '標準時間']) || '',
        unit: String(pickRowValue_(headers, row, ['数量単位', '単位']) || '').trim(),
        busyTarget: isBlank_(pickRowValue_(headers, row, ['繁忙期加算対象', '繁忙期対象']))
          ? false : parseBooleanLoose_(pickRowValue_(headers, row, ['繁忙期加算対象', '繁忙期対象'])),
        busySurchargeRaw: pickRowValue_(headers, row, ['繁忙期加算額']) || '',
        discountTarget: isBlank_(pickRowValue_(headers, row, ['割引対象']))
          ? false : parseBooleanLoose_(pickRowValue_(headers, row, ['割引対象'])),
        multipleDiscountTarget: parseBooleanLoose_(pickRowValue_(headers, row, ['複数台割引対象'])),
        invoiceReusable: isBlank_(pickRowValue_(headers, row, ['請求書流用', '請求書流用対象']))
          ? true : parseBooleanLoose_(pickRowValue_(headers, row, ['請求書流用', '請求書流用対象'])),
        sortOrder: toNumber_(pickRowValue_(headers, row, ['表示順'])),
        note: String(pickRowValue_(headers, row, ['補足_現場入力', '備考']) || ''),
        requireCheck: String(pickRowValue_(headers, row, ['要確認事項']) || '')
      };
    })
    .filter(function (m) { return m.menuId && m.name; })
    .filter(function (m) { return isActive_(pickMenuActiveValue_(sheet, headerInfo, m.menuId)); })
    .sort(function (a, b) { return (a.sortOrder || 9999) - (b.sortOrder || 9999); });
}

function pickMenuActiveValue_(sheet, headerInfo, menuId) {
  const lastRow = sheet.getLastRow();
  const lastCol = sheet.getLastColumn();
  const headers = sheet.getRange(headerInfo.headerRow, 1, 1, lastCol).getDisplayValues()[0]
    .map(function (h) { return String(h || '').trim(); });
  const values = sheet.getRange(headerInfo.headerRow + 1, 1, lastRow - headerInfo.headerRow, lastCol).getValues();

  for (let i = 0; i < values.length; i++) {
    if (String(pickRowValue_(headers, values[i], ['メニューID']) || '').trim() === menuId) {
      return pickRowValue_(headers, values[i], ['有効']);
    }
  }
  return true;
}

function normalizeMenuType_(menuId, rawMenuType) {
  const id = String(menuId || '');
  const raw = String(rawMenuType || '');
  if (id.indexOf('O') === 0) return 'オプション';
  if (raw === 'オプション' || raw === 'セット') return 'オプション';
  return 'メイン';
}

function pickRowValue_(headers, row, candidateHeaders) {
  for (let i = 0; i < candidateHeaders.length; i++) {
    for (let c = 0; c < headers.length; c++) {
      if (headers[c] === candidateHeaders[i] && !isBlank_(row[c])) return row[c];
    }
  }

  for (let i = 0; i < candidateHeaders.length; i++) {
    for (let c = 0; c < headers.length; c++) {
      if (headers[c] === candidateHeaders[i]) return row[c];
    }
  }

  return '';
}

function getCellDefMap_(docType, ctx) {
  const sheet = ensureSheetWithHeaders_(getMasterSs_(), APP.SHEET_CELL_DEF, getCellDefHeaders_());
  const headerInfo = findHeaderInfo_(sheet, ['帳票区分', '対象シート', '項目キー']);
  if (!headerInfo || !headerInfo.headerRow) return getDefaultEstimateCellDefMap_();

  const lastRow = sheet.getLastRow();
  const lastCol = sheet.getLastColumn();
  if (lastRow <= headerInfo.headerRow) return getDefaultEstimateCellDefMap_();

  const headers = sheet.getRange(headerInfo.headerRow, 1, 1, lastCol).getDisplayValues()[0]
    .map(function (h) { return String(h || '').trim(); });
  const values = sheet.getRange(headerInfo.headerRow + 1, 1, lastRow - headerInfo.headerRow, lastCol).getValues();
  const map = {};

  values.forEach(function (row) {
    const rowDocType = String(pickRowValue_(headers, row, ['帳票区分']) || '').trim();
    if (rowDocType !== docType) return;

    const key = String(pickRowValue_(headers, row, ['項目キー']) || '').trim();
    if (!key) return;

    const targetSheet = String(pickRowValue_(headers, row, ['対象シート']) || '').trim();
    const rawA1 = String(pickRowValue_(headers, row, ['セル/範囲']) || '').trim();
    const fixedA1 = isValidA1_(rawA1) ? rawA1 : findA1InRow_(row);
    if (!fixedA1) return;

    map[key] = { key: key, targetSheet: targetSheet, a1: fixedA1 };
  });

  return Object.keys(map).length ? map : getDefaultEstimateCellDefMap_();
}

function isValidA1_(s) {
  return /^[A-Z]+\d+(?::[A-Z]+\d+)?$/i.test(String(s || '').trim());
}

function findA1InRow_(row) {
  for (let i = 0; i < row.length; i++) {
    const value = String(row[i] || '').trim();
    if (isValidA1_(value)) return value;
  }
  return '';
}

function generateEstimateId_(sheet) {
  const today = formatDate_(new Date(), 'yyyyMMdd');
  const prefix = 'EST-' + today + '-';
  const rows = readObjects_(sheet, ['estimate_id']);
  let maxNo = 0;

  rows.forEach(function (r) {
    const id = String(r['estimate_id'] || '');
    if (id.indexOf(prefix) !== 0) return;
    const n = Number(id.replace(prefix, ''));
    if (!isNaN(n) && n > maxNo) maxNo = n;
  });

  return prefix + String(maxNo + 1).padStart(4, '0');
}

function updateEstimateFields_(estimateId, fields) {
  const sheet = ensureSheetWithHeaders_(getTemplateSs_(getSettings_()), APP.SHEET_ESTIMATE_DATA, getEstimateHeaders_());
  const found = findEstimateRecord_(sheet, estimateId);
  if (!found) throw new Error('更新対象の見積IDが見つかりません：' + estimateId);
  updateObjectAtRow_(sheet, found.rowNumber, fields);
}

function findEstimateRecord_(sheet, estimateId) {
  const info = findHeaderInfo_(sheet, ['estimate_id']);
  if (!info || !info.map['estimate_id']) throw new Error('見積データシートに estimate_id ヘッダーがありません。');

  const lastRow = sheet.getLastRow();
  if (lastRow <= info.headerRow) return null;

  const ids = sheet.getRange(info.headerRow + 1, info.map['estimate_id'], lastRow - info.headerRow, 1).getDisplayValues();

  for (let i = 0; i < ids.length; i++) {
    if (String(ids[i][0]).trim() === String(estimateId).trim()) {
      const rowNumber = info.headerRow + 1 + i;
      return { rowNumber: rowNumber, record: readObjectAtRow_(sheet, rowNumber, info) };
    }
  }

  return null;
}

function ensureSheetWithHeaders_(ss, sheetName, headers) {
  let sheet = ss.getSheetByName(sheetName);
  if (!sheet) sheet = ss.insertSheet(sheetName);

  if (sheet.getLastRow() === 0 || sheet.getLastColumn() === 0) {
    sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
    sheet.setFrozenRows(1);
    return sheet;
  }

  const info = findHeaderInfo_(sheet, headers.slice(0, Math.min(2, headers.length)));
  const headerRow = info ? info.headerRow : 1;
  const map = info ? info.map : {};
  const missing = headers.filter(function (h) { return !map[h]; });

  if (missing.length > 0) {
    sheet.getRange(headerRow, sheet.getLastColumn() + 1, 1, missing.length).setValues([missing]);
  }

  sheet.setFrozenRows(headerRow);
  return sheet;
}

function findHeaderInfo_(sheet, requiredHeaders) {
  const lastRow = Math.min(Math.max(sheet.getLastRow(), 1), 20);
  const lastCol = Math.max(sheet.getLastColumn(), 1);
  const values = sheet.getRange(1, 1, lastRow, lastCol).getDisplayValues();
  let best = null;

  for (let r = 0; r < values.length; r++) {
    const map = {};
    values[r].forEach(function (v, idx) {
      const key = String(v || '').trim();
      if (key && !map[key]) map[key] = idx + 1;
    });

    const count = (requiredHeaders || []).filter(function (h) { return map[h]; }).length;
    if (!best || count > best.count) best = { headerRow: r + 1, map: map, count: count };
  }

  return best;
}

function readObjects_(sheet, requiredHeaders) {
  const info = findHeaderInfo_(sheet, requiredHeaders || []);
  if (!info) return [];

  const lastRow = sheet.getLastRow();
  const lastCol = sheet.getLastColumn();
  if (lastRow <= info.headerRow) return [];

  const values = sheet.getRange(info.headerRow + 1, 1, lastRow - info.headerRow, lastCol).getValues();
  const headersByCol = {};
  Object.keys(info.map).forEach(function (h) { headersByCol[info.map[h]] = h; });

  return values
    .filter(function (row) { return row.some(function (v) { return !isBlank_(v); }); })
    .map(function (row) {
      const obj = {};
      Object.keys(headersByCol).forEach(function (colText) {
        const col = Number(colText);
        obj[headersByCol[col]] = row[col - 1];
      });
      return obj;
    });
}

function readObjectAtRow_(sheet, rowNumber, info) {
  const row = sheet.getRange(rowNumber, 1, 1, sheet.getLastColumn()).getValues()[0];
  const obj = {};
  Object.keys(info.map).forEach(function (h) { obj[h] = row[info.map[h] - 1]; });
  return obj;
}

function appendObject_(sheet, headers, obj) {
  ensureSheetWithHeaders_(sheet.getParent(), sheet.getName(), headers);
  sheet.appendRow(headers.map(function (h) { return obj[h] !== undefined ? obj[h] : ''; }));
}

function updateObjectAtRow_(sheet, rowNumber, fields) {
  const info = findHeaderInfo_(sheet, Object.keys(fields));
  const map = info.map;

  Object.keys(fields).forEach(function (key) {
    if (!map[key]) {
      const newCol = sheet.getLastColumn() + 1;
      sheet.getRange(info.headerRow, newCol).setValue(key);
      map[key] = newCol;
    }
    sheet.getRange(rowNumber, map[key]).setValue(fields[key]);
  });
}

function upsertRowsByKey_(sheet, keyHeader, rows, headers) {
  ensureSheetWithHeaders_(sheet.getParent(), sheet.getName(), headers);
  const existing = readObjects_(sheet, [keyHeader]);
  const exists = {};
  existing.forEach(function (r) { exists[String(r[keyHeader] || '').trim()] = true; });
  const keyIndex = headers.indexOf(keyHeader);

  const rowsToAdd = rows.filter(function (row) {
    return !exists[String(row[keyIndex] || '').trim()];
  });

  if (rowsToAdd.length > 0) {
    sheet.getRange(sheet.getLastRow() + 1, 1, rowsToAdd.length, headers.length).setValues(rowsToAdd);
  }
}

/**
 * 複数セル範囲へ同一値を書いて二重表示しないため、
 * 範囲全体をclearして左上セルだけに書き込む。
 */
function setByKey_(sheet, defs, key, value) {
  if (!defs[key] || !defs[key].a1) return;
  const range = sheet.getRange(defs[key].a1);
  range.clearContent();
  sheet.getRange(range.getRow(), range.getColumn()).setValue(value);
}

function getTargetSheetNameFromDefs_(defs, fallback) {
  const keys = Object.keys(defs);
  for (let i = 0; i < keys.length; i++) {
    if (defs[keys[i]].targetSheet) return defs[keys[i]].targetSheet;
  }
  return fallback;
}

function parseDateInput_(value) {
  if (!value) return null;
  if (Object.prototype.toString.call(value) === '[object Date]') return value;

  const s = String(value).trim();
  const m = s.match(/^(\d{4})-(\d{1,2})-(\d{1,2})$/);
  if (m) return new Date(Number(m[1]), Number(m[2]) - 1, Number(m[3]));

  const d = new Date(s);
  return isNaN(d.getTime()) ? null : d;
}

function toDateInputValue_(value) {
  const d = parseDateInput_(value);
  return d ? formatDate_(d, 'yyyy-MM-dd') : '';
}

function formatDate_(date, pattern) {
  return Utilities.formatDate(parseDateInput_(date) || new Date(), APP.TZ, pattern);
}

function formatDateTime_(date) {
  if (!date) return '';
  const d = parseDateInput_(date) || date;
  try { return Utilities.formatDate(d, APP.TZ, 'yyyy/MM/dd HH:mm:ss'); }
  catch (e) { return String(date); }
}

function toNumber_(value) {
  if (typeof value === 'number') return value;
  if (value === null || value === undefined) return 0;

  const s = String(value).replace(/[,\s円¥￥]/g, '').replace(/％/g, '%').trim();
  if (!s) return 0;
  if (s.indexOf('%') >= 0) return Number(s.replace('%', '')) / 100;

  const n = Number(s);
  return isNaN(n) ? 0 : n;
}

function normalizeRate_(value) {
  const n = toNumber_(value);
  return n > 1 ? n / 100 : n;
}

function getTaxRate_(settings) {
  const rate = normalizeRate_(settings['税率'] || 0.10);
  return rate || 0.10;
}

function roundYen_(n) { return Math.round(toNumber_(n)); }
function formatYen_(n) { return '¥' + roundYen_(n).toLocaleString('ja-JP'); }

function isTaxable_(taxType) {
  return !containsAny_(String(taxType || ''), ['非課税','不課税','対象外']);
}

function parseBooleanLoose_(v) {
  if (v === true) return true;
  if (v === false) return false;
  const s = String(v || '').trim().toLowerCase();
  return ['true','1','yes','y','有効','○','〇','on','適用'].indexOf(s) >= 0;
}

function isActive_(v) {
  if (isBlank_(v)) return true;
  const s = String(v).trim().toLowerCase();
  return ['false','0','no','n','無効','停止','off'].indexOf(s) < 0;
}

function boolText_(b) { return b ? 'TRUE' : 'FALSE'; }
function isBlank_(v) { return v === null || v === undefined || String(v).trim() === ''; }

function containsAny_(text, keywords) {
  const s = String(text || '');
  return keywords.some(function (k) { return s.indexOf(k) >= 0; });
}

function matchText_(value, keyword) {
  if (!keyword) return true;
  return String(value || '').toLowerCase().indexOf(String(keyword).toLowerCase()) >= 0;
}

function matchDate_(value, dateText) {
  if (!dateText) return true;
  return formatDate_(value, 'yyyy-MM-dd') === dateText;
}

function pad2_(n) { return String(n).padStart(2, '0'); }

function sanitizeFileName_(name) {
  return String(name || 'file.pdf').replace(/[\\/:*?"<>|#%{}~&]/g, '_');
}

function toErrorMessage_(e) { return !e ? '' : (e.message ? e.message : String(e)); }

function truncate_(s, max) {
  s = String(s || '');
  return s.length > max ? s.slice(0, max) + '...' : s;
}
